buat_matriks <- function(nama) {
  out <- matrix(NA_real_, length(bulan), nrow(grid),
                dimnames = list(as.character(bulan), grid$grid_id))
  i <- match(panel$month, bulan)
  j <- match(panel$grid_id, grid$grid_id)
  out[cbind(i, j)] <- panel[[nama]]
  out
}

Y <- log1p(buat_matriks("chla_mg_m3"))
X_sst <- buat_matriks("sst_c")
X_angin <- buat_matriks("wind_speed_m_s")

kor_lag <- function(x, jeda) {
  a <- x[seq_len(length(x) - jeda)]
  b <- x[seq.int(jeda + 1, length(x))]
  valid <- is.finite(a) & is.finite(b)
  if (sum(valid) < 36 || sd(a[valid]) == 0 || sd(b[valid]) == 0) return(NA_real_)
  cor(a[valid], b[valid])
}

8. Hubungan dengan SST dan angin

hapus_musim <- function(x) {
  m <- as.integer(format(bulan, "%m"))
  out <- x
  for (j in seq_len(ncol(x))) {
    for (k in 1:12) {
      idx <- which(m == k)
      rerata <- mean(x[idx, j], na.rm = TRUE)
      out[idx, j] <- x[idx, j] - rerata
    }
  }
  out
}

Y_anom <- hapus_musim(Y)
SST_anom <- hapus_musim(X_sst)
angin_anom <- hapus_musim(X_angin)

kor_prediktor <- function(y, x, jeda) {
  if (jeda > 0) {
    y <- y[seq.int(jeda + 1, nrow(y)), , drop = FALSE]
    x <- x[seq_len(nrow(x) - jeda), , drop = FALSE]
  }
  vapply(seq_len(ncol(y)), function(j) {
    valid <- is.finite(y[, j]) & is.finite(x[, j])
    if (sum(valid) < 36 || sd(y[valid, j]) == 0 ||
        sd(x[valid, j]) == 0) return(NA_real_)
    cor(y[valid, j], x[valid, j])
  }, numeric(1))
}

9. Ketergantungan spasial

dx <- abs(outer(grid$lon, grid$lon, "-"))
dy <- abs(outer(grid$lat, grid$lat, "-"))
tol <- 1e-6
W_rook <- (abs(dx - 0.25) < tol & dy < tol) |
          (abs(dy - 0.25) < tol & dx < tol)
W_queen <- dx <= 0.25 + tol & dy <= 0.25 + tol & (dx + dy) > tol
diag(W_rook) <- FALSE
diag(W_queen) <- FALSE

moran <- function(x, W) {
  idx <- which(is.finite(x))
  if (length(idx) < 3) return(NA_real_)
  w <- W[idx, idx, drop = FALSE]
  s0 <- sum(w)
  z <- x[idx] - mean(x[idx])
  penyebut <- sum(z^2)
  if (s0 == 0 || penyebut == 0) return(NA_real_)
  length(idx) * sum(w * outer(z, z)) / (s0 * penyebut)
}

phi <- rep(NA_real_, ncol(Y_anom))
residu <- matrix(NA_real_, nrow(Y_anom), ncol(Y_anom), dimnames = dimnames(Y_anom))
for (j in seq_len(ncol(Y_anom))) {
  a <- Y_anom[-nrow(Y_anom), j]
  b <- Y_anom[-1, j]
  ok <- is.finite(a) & is.finite(b)
  penyebut <- sum(a[ok]^2)
  if (sum(ok) < 36 || penyebut == 0) next
  phi[j] <- sum(a[ok] * b[ok]) / penyebut
  residu[-1, j] <- b - phi[j] * a
}

ringkas_moran <- function(W, nama) {
  indeks <- data.frame(
    month = bulan,
    raw = apply(Y, 1, moran, W = W),
    anomaly = apply(Y_anom, 1, moran, W = W),
    residual = apply(residu, 1, moran, W = W)
  )
  data.frame(
    Bobot = nama,
    Pasangan = sum(W) / 2,
    Grid_tanpa_tetangga = sum(rowSums(W) == 0),
    Median_mentah = median(indeks$raw, na.rm = TRUE),
    Median_tanpa_musim = median(indeks$anomaly, na.rm = TRUE),
    Median_residu_AR1 = median(indeks$residual, na.rm = TRUE)
  )
}

spasial <- rbind(ringkas_moran(W_rook, "Rook"),
                 ringkas_moran(W_queen, "Queen"))
spasial[, 4:6] <- round(spasial[, 4:6], 3)
